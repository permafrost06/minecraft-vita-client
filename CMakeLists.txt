cmake_minimum_required(VERSION 3.20)
option(VITA "Build PlayStation Vita application" OFF)
if(VITA AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
endif()
project(VitaBlockClient C)
set(CMAKE_C_STANDARD 11)
include(FetchContent)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zlib URL https://zlib.net/fossils/zlib-1.3.1.tar.gz
  URL_HASH SHA256=9a93b2b7dfdac77ceba5a558a580e74667dd6fede4585b91eefb60f03b72df23
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(zlib)
add_library(mcnet src/net.c src/protocol.c src/reader.c src/nbt.c src/world.c src/render.c src/player.c src/mesh.c src/textures.c src/interaction.c src/entities.c src/entity_mesh.c)
target_include_directories(mcnet PUBLIC src PRIVATE ${zlib_SOURCE_DIR} ${zlib_BINARY_DIR})
target_link_libraries(mcnet PUBLIC zlibstatic m)
target_compile_options(mcnet PRIVATE -O2 -Wall -Wextra -Werror)
if(VITA)
  include("${VITASDK}/share/vita.cmake" REQUIRED)
  target_compile_definitions(mcnet PUBLIC __vita__)
  target_link_libraries(mcnet PUBLIC SceNet_stub SceSysmodule_stub)
  add_executable(vitablock src/main_vita.c src/gl_render.c)
  target_include_directories(vitablock PRIVATE vendor)
  target_compile_options(vitablock PRIVATE -O2 -Wall -Wextra)
  target_link_libraries(vitablock mcnet vitaGL vitashark SceShaccCgExt mathneon taihen_stub SceShaccCg_stub
    SceKernelDmacMgr_stub SceGxm_stub SceDisplay_stub ScePower_stub SceCtrl_stub SceCommonDialog_stub SceAppMgr_stub
    SceIofilemgr_stub SceLibKernel_stub stdc++ m)
  vita_create_self(eboot.bin vitablock UNSAFE)
  vita_create_vpk(VitaBlockClient.vpk VBLK00001 eboot.bin VERSION 00.01 NAME "Vita Block Client")
else()
  add_executable(mcprobe src/main_host.c)
  target_link_libraries(mcprobe mcnet)
  enable_testing()
  add_executable(protocol_tests tests/protocol_tests.c)
  target_link_libraries(protocol_tests mcnet)
  add_test(NAME protocol COMMAND protocol_tests)
  add_executable(world_tests tests/world_tests.c)
  target_link_libraries(world_tests mcnet)
  add_test(NAME world COMMAND world_tests)
  add_executable(player_tests tests/player_tests.c)
  target_link_libraries(player_tests mcnet)
  add_test(NAME player COMMAND player_tests)
  add_executable(mesh_tests tests/mesh_tests.c)
  target_link_libraries(mesh_tests mcnet)
  add_test(NAME mesh COMMAND mesh_tests)
  add_executable(movement_client tests/movement_client.c)
  target_link_libraries(movement_client mcnet)
  add_executable(interaction_tests tests/interaction_tests.c)
  target_link_libraries(interaction_tests mcnet)
  add_test(NAME interaction COMMAND interaction_tests)
  add_executable(interaction_client tests/interaction_client.c)
  target_link_libraries(interaction_client mcnet)
  add_executable(live_interaction tests/live_interaction.c)
  target_link_libraries(live_interaction mcnet)
  add_executable(entity_actor tests/entity_actor.c)
  target_link_libraries(entity_actor mcnet)
  add_executable(entity_tests tests/entity_tests.c)
  target_link_libraries(entity_tests mcnet)
  add_test(NAME entity COMMAND entity_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME tcp_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:mcprobe> $<TARGET_FILE:movement_client> $<TARGET_FILE:interaction_client>)
endif()
